Thiết kế worker pool trong Go để import 50.000 review cho một shop: giới hạn concurrency, xử lý lỗi, dừng khi bị cancel.

Kiến thức
Xem đáp án

Dùng errgroup với SetLimit để giới hạn số goroutine. Lỗi từng item thì ghi lại, không hủy cả job; chỉ dừng khi context bị cancel. Ghi DB theo batch thay vì từng dòng.

func (s *Importer) Import(ctx context.Context, shopID int64, items []RawReview) (Result, error) {
    g, ctx := errgroup.WithContext(ctx)
    g.SetLimit(16) // giới hạn song song

    out := make(chan Review, 256)
    var failed atomic.Int64

    // Producer: normalize song song
    go func() {
        defer close(out)
        for _, it := range items {
            it := it
            g.Go(func() error {
                r, err := s.normalize(ctx, shopID, it)
                if err != nil {
                    failed.Add(1) // lỗi từng item không làm hỏng cả job
                    return nil
                }
                select {
                case out <- r:
                case <-ctx.Done():
                    return ctx.Err()
                }
                return nil
            })
        }
        _ = g.Wait()
    }()

    // Consumer: ghi DB theo batch 500
    batch := make([]Review, 0, 500)
    var saved int
    for r := range out {
        batch = append(batch, r)
        if len(batch) == cap(batch) {
            if err := s.repo.UpsertBatch(ctx, batch); err != nil { return Result{}, err }
            saved += len(batch)
            batch = batch[:0]
        }
    }
    if len(batch) > 0 {
        if err := s.repo.UpsertBatch(ctx, batch); err != nil { return Result{}, err }
        saved += len(batch)
    }
    return Result{Saved: saved, Failed: int(failed.Load())}, ctx.Err()
}
Trade-off: job lớn nên chạy ở worker riêng qua queue (không trong request HTTP), lưu tiến độ để resume và hiển thị % cho merchant. Concurrency phụ thuộc vào thứ chặn bạn: DB (giới hạn bởi pool) hay nguồn bên ngoài (giới hạn bởi rate limit). Đặt riêng cho từng loại.

Node.js event loop hoạt động thế nào? Vì sao một hàm parse JSON/HTML nặng có thể làm "đứng" cả server? So với Go thì sao?

Kiến thức
Xem đáp án
  • Node chạy JS trên một thread; I/O (network, file) được libuv xử lý bất đồng bộ, callback được đưa lại vào event loop. Các phase: timers → pending callbacks → poll (I/O) → check (setImmediate) → close. Microtask (Promise) chạy sau mỗi callback.
  • Code CPU-bound (parse HTML lớn bằng cheerio, JSON.parse file vài chục MB, regex nặng) chiếm thread chính → mọi request khác phải chờ. Giải pháp: worker_threads, tách sang service/worker riêng, stream parsing, hoặc chia nhỏ việc.
  • Go: goroutine được scheduler phân phối lên nhiều OS thread (GOMAXPROCS), có preemption, nên một goroutine CPU-bound không chặn các request khác. Đây là lý do Go phù hợp cho worker parse/normalize dữ liệu.
Trade-off: Node mạnh ở I/O-bound, chia sẻ code/type với frontend React (TypeScript), hệ sinh thái Shopify (Shopify CLI, @shopify/shopify-api, template Remix/React Router) rất tốt. Go mạnh ở CPU-bound, concurrency, memory ổn định, binary nhỏ. Kiến trúc hợp lý: app Shopify (OAuth, admin UI) bằng Node theo template chính thức; pipeline crawl/ETL và API đọc tải cao bằng Go.

Context, timeout và cancel khi gọi API bên ngoài (Shopify, Etsy, Google) trong Go.

Kiến thức
Xem đáp án
  • Không dùng http.DefaultClient (không có timeout). Tạo client riêng cho từng nguồn với Timeout, Transport có MaxIdleConnsPerHost, IdleConnTimeout.
  • Mỗi call dùng http.NewRequestWithContext với ctx có deadline; luôn defer resp.Body.Close() và đọc hết body (để connection được tái sử dụng).
  • Giới hạn kích thước body đọc vào (io.LimitReader) để một response bất thường không làm tràn memory.
var shopifyClient = &http.Client{
    Timeout: 20 * time.Second,
    Transport: &http.Transport{
        MaxIdleConns:        200,
        MaxIdleConnsPerHost: 50,
        IdleConnTimeout:     90 * time.Second,
    },
}

func getJSON(ctx context.Context, url string, v any) error {
    req, _ := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
    resp, err := shopifyClient.Do(req)
    if err != nil { return err }
    defer resp.Body.Close()
    if resp.StatusCode >= 300 {
        io.Copy(io.Discard, io.LimitReader(resp.Body, 1<<16))
        return &HTTPError{Status: resp.StatusCode, RetryAfter: resp.Header.Get("Retry-After")}
    }
    return json.NewDecoder(io.LimitReader(resp.Body, 10<<20)).Decode(v)
}

Rate limit gọi ra ngoài: mỗi shop có quota Shopify riêng, mỗi nguồn crawl có giới hạn riêng. Thiết kế limiter thế nào?

Kiến thức
Xem đáp án
  • Một process: golang.org/x/time/rate (token bucket), map limiter theo key (shop_domain hoặc source_host), dọn key không dùng.
  • Nhiều worker/pod: limiter phân tán trên Redis (Lua script token bucket hoặc sliding window), vì quota là của shop chứ không phải của pod.
  • Thích ứng (adaptive): đọc header/phản hồi của nguồn (Shopify trả throttleStatus trong GraphQL, Retry-After khi 429) để tự giảm tốc thay vì cố định.
type Limiters struct {
    mu sync.Mutex
    m  map[string]*rate.Limiter
}

func (l *Limiters) Wait(ctx context.Context, key string, rps float64, burst int) error {
    l.mu.Lock()
    lim, ok := l.m[key]
    if !ok {
        lim = rate.NewLimiter(rate.Limit(rps), burst)
        l.m[key] = lim
    }
    l.mu.Unlock()
    return lim.Wait(ctx) // chặn tới khi có token hoặc ctx hết hạn
}
Trade-off: limiter local đơn giản, không phụ thuộc Redis, nhưng N pod sẽ gọi gấp N lần quota. Redis limiter đúng toàn cụm nhưng thêm 1 round-trip mỗi call. Cách lai: phân shard theo shop (cùng shop luôn vào cùng worker, ví dụ Kafka partition key = shop_id) để limiter local vẫn đúng.

Graceful shutdown cho worker đang xử lý job import/crawl khi deploy.

Kiến thức
Xem đáp án
  • Nhận SIGTERM → ngừng nhận job mới → cho job đang chạy hoàn thành phần hiện tại (hoặc lưu checkpoint) → ack/commit → thoát. Đặt terminationGracePeriodSeconds đủ lớn.
  • Job dài (import 50k review) phải resume được: lưu cursor/page đã xử lý vào DB. Nếu bị kill, job được giao lại và chạy tiếp từ checkpoint, không làm lại từ đầu.
  • Ghi idempotent (upsert theo khóa unique) nên xử lý lặp một phần cũng không tạo bản trùng.
ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGTERM, os.Interrupt)
defer stop()
for {
    job, err := queue.Next(ctx) // trả lỗi khi ctx bị cancel
    if err != nil { break }
    // dùng ctx riêng có hạn để job kịp lưu checkpoint
    jctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 25*time.Second)
    runWithCheckpoint(jctx, job)
    cancel()
}

Widget hiển thị review: sort theo "pinned trước, mới nhất", lọc theo số sao / có ảnh, phân trang. Viết query và chọn index.

Kiến thức
Xem đáp án
-- Trang đầu
SELECT id, rating, author, body, has_media, reviewed_at
FROM reviews
WHERE shop_id = ? AND product_id = ? AND status = 1
ORDER BY is_pinned DESC, reviewed_at DESC, id DESC
LIMIT 10;

-- Trang tiếp theo: keyset cursor (is_pinned, reviewed_at, id) của dòng cuối
WHERE shop_id = ? AND product_id = ? AND status = 1
  AND (is_pinned < ?
       OR (is_pinned = ? AND reviewed_at < ?)
       OR (is_pinned = ? AND reviewed_at = ? AND id < ?))
  • Index (shop_id, product_id, status, is_pinned, reviewed_at, id) khớp đúng thứ tự WHERE bằng (=) rồi ORDER BY, nên MySQL đọc theo index (không filesort) và dừng sau 10 dòng. MySQL 8 hỗ trợ index giảm dần (DESC) nếu cần trộn chiều sort.
  • Lọc theo sao (rating = 5) chen giữa sẽ phá thứ tự index. Có hai cách: thêm index riêng (shop_id, product_id, status, rating, reviewed_at, id), hoặc chấp nhận quét thêm vì mỗi sản phẩm thường chỉ vài trăm review.
  • Số review theo từng mức sao lấy từ product_rating_stats, không COUNT(*) mỗi lần.
Trade-off: keyset nhanh và ổn định nhưng không nhảy tới "trang 7" được. Widget review thường dùng "Xem thêm" (load more), nên keyset là hợp lý. Trang quản trị cần số trang thì dùng offset có giới hạn.

Import 50.000 review và tránh trùng khi merchant import lại cùng sản phẩm. Bulk insert/upsert thế nào?

Kiến thức
Xem đáp án
  • Khóa dedup: (shop_id, product_id, source, external_id) nếu nguồn có ID, nếu không thì hash nội dung đã chuẩn hoá (trim, lowercase, gộp khoảng trắng) + tác giả + ngày.
  • Batch insert 500–1000 dòng/câu lệnh (giới hạn bởi max_allowed_packet và số placeholder 65.535 của MySQL).
  • INSERT ... ON DUPLICATE KEY UPDATE cập nhật những field có thể đổi (rating/body) nhưng không đè trạng thái merchant đã chỉnh (hidden, pinned).
  • Sau batch: cập nhật lại product_rating_stats bằng 1 câu aggregate cho các product bị ảnh hưởng.
func (r *Repo) UpsertBatch(ctx context.Context, rows []Review) error {
    if len(rows) == 0 { return nil }
    var sb strings.Builder
    args := make([]any, 0, len(rows)*9)
    sb.WriteString(`INSERT INTO reviews
      (shop_id, product_id, source, content_hash, rating, author, body, reviewed_at, status, created_at) VALUES `)
    for i, x := range rows {
        if i > 0 { sb.WriteByte(',') }
        sb.WriteString("(?,?,?,?,?,?,?,?,?,UTC_TIMESTAMP())")
        args = append(args, x.ShopID, x.ProductID, x.Source, x.Hash[:], x.Rating, x.Author, x.Body, x.ReviewedAt, x.Status)
    }
    sb.WriteString(` ON DUPLICATE KEY UPDATE rating = VALUES(rating), body = VALUES(body)`) // giữ nguyên status
    _, err := r.db.ExecContext(ctx, sb.String(), args...)
    return err
}

Ghi chú: MySQL 8.0.20+ khuyến nghị cú pháp alias (INSERT ... AS new ON DUPLICATE KEY UPDATE rating = new.rating) thay cho VALUES() đã deprecated.

Caching cho widget review: cache gì, ở tầng nào, invalidate thế nào?

Kiến thức
Xem đáp án

Widget là tải lớn nhất (mọi lượt xem trang sản phẩm của mọi shop). Dùng nhiều tầng cache:

Cache stampede / thundering herd khi key của một shop rất lớn hết hạn. Xử lý thế nào?

Kiến thức
Xem đáp án
  • singleflight trong mỗi process: nhiều request cùng key chỉ gọi DB 1 lần.
  • Lock ngắn trên Redis (SET key NX PX 3000) để chỉ 1 pod rebuild, các pod khác trả bản cũ (stale) hoặc chờ ngắn.
  • Stale-while-revalidate: lưu kèm thời điểm "mềm" hết hạn; quá hạn mềm thì vẫn trả dữ liệu cũ và refresh nền.

func (s *Svc) GetReviews(ctx context.Context, key string, load func() ([]byte, error)) ([]byte, error) {
    if b, err := s.rdb.Get(ctx, key).Bytes(); err == nil {
        return b, nil
    }
    v, err, _ := sf.Do(key, func() (any, error) {
        b, err := load()
        if err == nil {
            ttl := 300*time.Second + time.Duration(rand.Int64N(30))*time.Second // jitter
            s.rdb.Set(context.WithoutCancel(ctx), key, b, ttl)
        }
        return b, err
    })
    if err != nil { return nil, err }
    return v.([]byte), nil
}

Redis vs Memcached. Và những cấu trúc Redis nào hữu ích cho sản phẩm này?

Kiến thức
Xem đáp án
  • Memcached: chỉ key-value, đa luồng, đơn giản, không persistence; tốt cho cache thuần.
  • Redis: nhiều cấu trúc dữ liệu, persistence (RDB/AOF), replication, Lua script, pub/sub, stream. Gần như luôn được chọn vì dùng được cho nhiều việc hơn cache.

    • String/Hash: cache JSON widget, cấu hình shop.
    • Sorted set: hàng đợi job theo thời điểm chạy (gửi email xin review sau N ngày), rate limit sliding window.
    • Set / Bloom filter (RedisBloom): kiểm tra nhanh review đã crawl chưa.
    • HyperLogLog: đếm số lượt xem widget không trùng (unique visitor) với bộ nhớ cố định.
    • Stream: hàng đợi nhẹ khi chưa cần Kafka.
Lưu ý vận hành: đặt maxmemory-policy allkeys-lru cho Redis làm cache; tách instance cache với instance dùng làm queue/lock (vì không được để eviction xoá job).

// Verify hmac trên query của OAuth callback
func verifyOAuthHMAC(q url.Values, secret string) bool {
    got := q.Get("hmac")
    keys := make([]string, 0, len(q))
    for k := range q { if k != "hmac" && k != "signature" { keys = append(keys, k) } }
    sort.Strings(keys)
    parts := make([]string, len(keys))
    for i, k := range keys { parts[i] = k + "=" + q.Get(k) }
    mac := hmac.New(sha256.New, []byte(secret))
    mac.Write([]byte(strings.Join(parts, "&")))
    return hmac.Equal([]byte(hex.EncodeToString(mac.Sum(nil))), []byte(got))
}

func (h *Handler) ShopifyWebhook(c *gin.Context) {
    body, err := io.ReadAll(io.LimitReader(c.Request.Body, 5<<20))
    if err != nil { c.Status(400); return }

    mac := hmac.New(sha256.New, h.secret)
    mac.Write(body)
    expected := base64.StdEncoding.EncodeToString(mac.Sum(nil))
    if !hmac.Equal([]byte(expected), []byte(c.GetHeader("X-Shopify-Hmac-Sha256"))) {
        c.Status(401); return
    }

    evt := WebhookEvent{
        ID:    c.GetHeader("X-Shopify-Event-Id"),
        Topic: c.GetHeader("X-Shopify-Topic"),
        Shop:  c.GetHeader("X-Shopify-Shop-Domain"),
        Body:  body,
    }
    if err := h.queue.Publish(c.Request.Context(), evt); err != nil {
        c.Status(500); return // để Shopify retry
    }
    c.Status(200)
}

func streamJSONL(ctx context.Context, r io.Reader, handle func(json.RawMessage) error) error {
    sc := bufio.NewScanner(r)
    sc.Buffer(make([]byte, 0, 1<<20), 16<<20) // dòng có thể dài
    for sc.Scan() {
        if err := ctx.Err(); err != nil { return err }
        if err := handle(json.RawMessage(sc.Bytes())); err != nil { return err }

}

func verifyProxy(q url.Values, secret string) bool {
    keys := make([]string, 0, len(q))
    for k := range q { if k != "signature" { keys = append(keys, k) } }
    sort.Strings(keys)
    var sb strings.Builder
    for _, k := range keys { sb.WriteString(k + "=" + strings.Join(q[k], ",")) }
    mac := hmac.New(sha256.New, []byte(secret))
    mac.Write([]byte(sb.String()))
    return hmac.Equal([]byte(hex.EncodeToString(mac.Sum(nil))), []byte(q.Get("signature")))
}

func Variant(expID, visitorID string, weights []int) int {
    h := fnv.New32a()
    h.Write([]byte(expID + ":" + visitorID))
    bucket := int(h.Sum32() % 100)
    acc := 0
    for i, w := range weights { // ví dụ [50, 50] hoặc [90, 10]
        acc += w
        if bucket < acc { return i }
    }
    return 0
}
